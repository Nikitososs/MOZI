Листинг: Поиск делителя методом rho Полларда (`lab_5/rsa_attack.py`)

```python
from math import gcd

def pollard_rho_factor(n: int) -> tuple[int, int]:
    """Найти нетривиальный делитель n циклическим методом Полларда rho."""
    if n % 2 == 0:
        return 2, 1
    if n % 3 == 0:
        return 3, 1

    total_iterations = 0
    constant = 1
    while True:
        x = y = 2
        divisor = 1
        while divisor == 1:
            x = (x * x + constant) % n
            y = (y * y + constant) % n
            y = (y * y + constant) % n
            divisor = gcd(abs(x - y), n)
            total_iterations += 1
        if divisor != n:
            return divisor, total_iterations
        constant += 1
```
